"""Frozen-parameter controlled interventions; labels never enter recovery maps.

Exact qualified writes/resets are supplied uniformly. Protected archive access
belongs ONLY to its named control. Round clocks are distinct from event clocks.
Pure reads do not update state. Identical fixed points can be fast-forwarded
only in autonomous intervals; executed and logical updates are both recorded.
"""
from __future__ import annotations
import time,json
from pathlib import Path
import numpy as np
import torch
from safetensors.torch import load_file
from .models import Representation,Candidate,Repair,FastSlow,MODES,UNKNOWN
from .data import labels,fault,fault_matrix
from .util import setup,write_json,sha,array_digest,digest_tensors,environment

ROUNDS=(0,1,2,4,8,16,32,128)

def load_candidates(root,models,rep_seed,learn_seed):
    root,models=Path(root),Path(models)
    rep=Representation(root/'reference'/f'soft_slots_wide_seed{rep_seed}.safetensors')
    learned={}
    for kind,cls in [('readout_repair',Repair),('recurrent',Repair),('fast_slow',FastSlow)]:
        m=cls();p=models/f'{kind}_rep{rep_seed}_seed{learn_seed}.safetensors'
        m.load_state_dict(load_file(str(p)),strict=True);m.eval();m.requires_grad_(False);learned[kind]=m
    candidates={}
    for mode in MODES:
        repair=learned['readout_repair'] if mode=='readout_repair' else learned['recurrent']
        candidates[mode]=Candidate(rep,mode,repair,learned['fast_slow'])
    return rep,learned,candidates

def iterate(c,state,rounds):
    """Yield (round,state,last_delta,actually_executed). Autonomous interval."""
    wanted=set(rounds);done=0
    if 0 in wanted:yield 0,state,None,0
    maximum=max(rounds)
    for k in range(1,maximum+1):
        before=state;state=c.step(state);done+=1
        if any(not torch.isfinite(x).all() for x in state):raise FloatingPointError('Nonfinite state')
        delta=max(float((x-y).abs().max()) for x,y in zip(state,before))
        if k in wanted:yield k,state,delta,done
        if all(torch.equal(x,y) for x,y in zip(state,before)):
            for future in sorted(wanted):
                if future>k:yield future,state,0.,done
            break


def metric(c,state,target,base_prediction=None,delta=None):
    pred=c.predict(state,archive=target if c.mode=='exact_archive' else None)
    view=c.view(state)
    correct=pred==target;known=target!=UNKNOWN;unknown=~known
    def count(mask):return dict(n=int(mask.sum()),correct=int((correct&mask).sum()),
                               accuracy=float(correct[mask].float().mean()) if bool(mask.any()) else None)
    m=dict(all=count(torch.ones_like(known)),known=count(known),unknown=count(unknown),
           latent_mse=float((view-c.rep.codes[target]).square().mean()),
           stored_fast_mse=float((state[0]-c.rep.codes[target]).square().mean()),
           max_state_abs=max(float(x.abs().max()) for x in state),
           prediction_sha256=array_digest(pred),target_sha256=array_digest(target),
           last_step_linf=delta,
           stable_wrong_count=int((~correct).sum()) if delta is not None and delta<1e-5 else 0)
    if base_prediction is not None:
        was=base_prediction==target
        m.update(recovered_count=int((correct&~was).sum()),harmed_count=int((~correct&was).sum()),
                 initially_wrong=int((~was).sum()))
    return m,pred

@torch.no_grad()
def evaluate(root,models,output,rep_seed,learn_seed,development=False):
    setup();root,models,output=Path(root),Path(models),Path(output)
    output.mkdir(parents=True,exist_ok=True)
    path=output/f'rep{rep_seed}_seed{learn_seed}.json'
    if path.exists():raise FileExistsError('No overwrite of completed evaluation')
    rep,learned,candidates=load_candidates(root,models,rep_seed,learn_seed)
    identities={k:digest_tensors(v.state_dict()) for k,v in learned.items()}
    before_rep=digest_tensors(rep.state_dict())
    rows=[];predictions={};t0=time.perf_counter()
    def add(group,condition,level,rounds,c,state,target,base=None,delta=None,executed=0,extra=None):
        m,pred=metric(c,state,target,base,delta)
        row=dict(group=group,condition=condition,level=level,rounds=rounds,mode=c.mode,
                 rep_seed=rep_seed,learn_seed=learn_seed,executed_autonomous_updates=executed,**m)
        if extra:row.update(extra)
        rows.append(row);predictions[f'p{len(rows)-1:05d}']=pred.numpy().astype(np.uint8)
    # Inputs independent of training generators and other main reference pairs.
    g=torch.Generator().manual_seed((3_000_000 if not development else 9_000_000)+learn_seed)
    y=labels(g,256 if not development else 64)
    faults=fault_matrix() if not development else [('clean',0.),('gaussian',.5),('gaussian',1.),('erase_all',1.)]
    for fi,(kind,level) in enumerate(faults):
        for mode,c in candidates.items():
            fg=torch.Generator().manual_seed(4_000_000+learn_seed*100+fi)
            state=fault(c.state(y),kind,level,fg,rep.codes,y)
            based=c.predict(state,archive=y if mode=='exact_archive' else None)
            input_digest=array_digest(*state)
            for k,s,delta,executed in iterate(c,state,ROUNDS):
                add('shock',kind,level,k,c,s,y,based,delta,executed,dict(input_state_sha256=input_digest))
    if not development:
        # Long autonomous clean maintenance, explicitly distinct from reasoning.
        yg=labels(torch.Generator().manual_seed(5_000_000+learn_seed),64)
        for c in candidates.values():
            for k,s,delta,executed in iterate(c,c.state(yg),(0,16,256,4096)):
                add('long_clean','no_new_observation',0.,k,c,s,yg,delta=delta,executed=executed)
        # One genuinely new qualified event at address0; other addresses unchanged.
        for fi,(kind,level) in enumerate([('clean',0.),('gaussian',.5),('toward_wrong',.75)]):
            new=y.clone();new[:,0]=(y[:,0]+1).remainder(8)
            mask=torch.zeros_like(y,dtype=torch.bool);mask[:,0]=True
            for c in candidates.values():
                fg=torch.Generator().manual_seed(6_000_000+learn_seed*100+fi)
                s=fault(c.state(y),kind,level,fg,rep.codes,y)
                s=list(iterate(c,s,(32,)))[-1][1]
                s=c.write(s,new,mask)
                base=c.predict(s,archive=new if c.mode=='exact_archive' else None)
                for k,state,delta,executed in iterate(c,s,(0,1,4,32,128)):
                    pred=c.predict(state,archive=new if c.mode=='exact_archive' else None)
                    add('overwrite',kind,level,k,c,state,new,base,delta,executed,
                        dict(changed_address_n=int(mask.sum()),changed_address_correct=int(((pred==new)&mask).sum()),
                             stale_old_value_count=int(((pred==y)&mask).sum())))
        # Explicit reset must clear both lanes and not hallucinate after maintenance.
        zeros=torch.full_like(y,UNKNOWN)
        for c in candidates.values():
            s=c.reset(c.state(y))
            for k,state,delta,executed in iterate(c,s,(0,1,4,32,4096)):
                add('reset','explicit_clear',0.,k,c,state,zeros,delta=delta,executed=executed)
        # Repeated state damage: independent Gaussian per lane and per event.
        # All candidates receive identical first-lane perturbations. Dual copies
        # receive an independent second-lane perturbation; replication is counted.
        yc=labels(torch.Generator().manual_seed(7_000_000+learn_seed),64)
        checkpoints=(1,16,64,256,1024)
        for ci,sigma in enumerate((.05,.15)):
            for c in candidates.values():
                ng=torch.Generator().manual_seed(8_000_000+learn_seed*10+ci)
                s=c.state(yc)
                for event in range(1,max(checkpoints)+1):
                    n1=torch.randn(s[0].shape,generator=ng)*sigma
                    n2=torch.randn(s[0].shape,generator=ng)*sigma
                    s=tuple(x+n for x,n in zip(s,(n1,n2)))
                    s=c.step(s)
                    if event in checkpoints:
                        add('chronic','independent_state_noise',sigma,event,c,s,yc,
                            extra=dict(maintenance_updates_per_event=1,logical_events=event))
            # Equal repair-call budget: hold all observations of damaged state,
            # run the same recurrent map only at the end. No extra external data.
            ng=torch.Generator().manual_seed(8_000_000+learn_seed*10+ci)
            z=rep.codes[yc].clone();c=candidates['recurrent']
            for event in range(1,max(checkpoints)+1):
                n1=torch.randn(z.shape,generator=ng)*sigma
                _=torch.randn(z.shape,generator=ng)*sigma
                z=z+n1
                if event in (16,64,256,1024):
                    ss=list(iterate(c,(z.clone(),),(event,)))[-1]
                    k,state,delta,executed=ss
                    add('chronic_endpoint','all_repair_at_end',sigma,event,c,state,yc,delta=delta,executed=executed,
                        extra=dict(maintenance_updates_per_event=0,logical_events=event,equal_logical_repair_budget=True))
    if before_rep!=digest_tensors(rep.state_dict()):raise RuntimeError('Source representation changed')
    if identities!={k:digest_tensors(v.state_dict()) for k,v in learned.items()}:raise RuntimeError('Evaluation changed weights')
    npz=output/(path.stem+'.predictions.npz');np.savez_compressed(npz,**predictions)
    result=dict(status='RUN',rep_seed=rep_seed,learn_seed=learn_seed,development=development,
                rows=rows,cells=len(rows),frozen_weights_unchanged=True,reference_sha256=rep.source_sha256,
                trained_tensor_identities=identities,predictions_file=npz.name,predictions_sha256=sha(npz),
                eval_wall_seconds=time.perf_counter()-t0,environment=environment(),
                scope='Qualified-write controlled maintenance, reused learned representations; not whole-model or brain simulation',
                exact_archive_note='Protected external last-record values; extra intact evidence, not a neural recovery oracle available to other modes')
    write_json(path,result)
    return {k:v for k,v in result.items() if k!='rows'}
