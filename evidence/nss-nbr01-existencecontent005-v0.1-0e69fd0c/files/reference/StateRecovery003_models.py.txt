"""Fixed learned representation, exact qualified writes, candidate maintenance.

The eight vectors and original readout are reused from HorizonBinding002.
No encoder, source gate, or source readout is trained here. For this controlled
study the supplied clean score rule qualifies writes; the learned source gate
is not used. Thus this is a qualified-write ablation, NOT an unchanged rerun
of the previous end-to-end learner. Addresses are explicitly provided.
"""
from __future__ import annotations
from pathlib import Path
import torch
from torch import nn
from safetensors.torch import load_file
from .util import sha
D=16
KEYS=4
UNKNOWN=8

class Representation(nn.Module):
    def __init__(self, checkpoint):
        super().__init__()
        src=load_file(str(checkpoint),device='cpu')
        self.head=nn.Sequential(nn.LayerNorm(D), nn.Linear(D,32), nn.GELU(),nn.Linear(32,9))
        self.head.load_state_dict({k[5:]:v for k,v in src.items() if k.startswith('head.')},strict=True)
        value=src['value.weight']
        scale=value.square().mean().sqrt()
        self.register_buffer('scale',scale)
        self.register_buffer('codes',torch.cat((value/scale,torch.zeros(1,D)),0))
        self.source_sha256=sha(checkpoint)
        self.requires_grad_(False);self.eval()
    def forward(self,z): return self.head(z*self.scale)
    def nearest(self,z):
        return ((z.unsqueeze(-2)-self.codes)**2).sum(-1).argmin(-1)

class Repair(nn.Module):
    """Learned residual denoising map. F(0)=0 by construction; not a proof
    of global contraction or biological fidelity. Reuse across maintenance
    rounds is actual recurrent state processing, not a chat-prompt loop.
    """
    def __init__(self):
        super().__init__()
        self.net=nn.Sequential(nn.Linear(D,64),nn.Tanh(),nn.Linear(64,D))
    def forward(self,z):
        baseline=self.net(torch.zeros(D,dtype=z.dtype,device=z.device))
        result=.5*z+.5*(self.net(z)-baseline)
        # GEMV/GEMM reduction orders can leave a tiny nonzero difference at
        # the mathematically reserved zero. Enforce the exact reset invariant;
        # nonzero damaged UNKNOWN state is NOT protected by this guard.
        return torch.where((z==0).all(-1,keepdim=True),torch.zeros_like(result),result)

class FastSlow(nn.Module):
    """Two coupled activation lanes, not mutable parameter associations.
    f'=(1-b)f+b*s; s'=(1-g)s+g*f (simultaneous old-state update).
    Qualified write: f=code; s=(1-a)s+a*code. Explicit reset clears both.
    All coefficients lie in (0,1). Unequal rates are a candidate, not presumed
    beneficial. There is no hidden third clean copy or event history.
    """
    def __init__(self):
        super().__init__()
        self.raw=nn.Parameter(torch.tensor([2.,0.,-3.]))
    def rates(self): return torch.sigmoid(self.raw)
    def forward(self,f,s,equal=False):
        a,b,g=self.rates()
        if equal: b=g=(b+g)/2
        return (1-b)*f+b*s,(1-g)*s+g*f
    def write(self,f,s,code,mask):
        a,_,_=self.rates()
        return torch.where(mask[...,None],code,f),torch.where(mask[...,None],(1-a)*s+a*code,s)

MODES=('hold','readout_repair','recurrent','recurrent_off','recurrent_query4',
       'fast_slow','fast_slow_off','fast_slow_equal','two_copy_mean',
       'nearest_read','nearest_maintenance','exact_archive','state_off')

class Candidate:
    def __init__(self,rep,mode,repair=None,fastslow=None):
        if mode not in MODES: raise ValueError('Unknown mode')
        if mode in ('readout_repair','recurrent','recurrent_off','recurrent_query4') and repair is None:
            raise ValueError('Repair checkpoint required')
        if mode.startswith('fast_slow') and fastslow is None: raise ValueError('Fast/slow checkpoint required')
        self.rep,self.mode,self.repair,self.fastslow=rep,mode,repair,fastslow
        self.dual=mode.startswith('fast_slow') or mode=='two_copy_mean'
        self.archive_values=None
    def state(self,y):
        x=self.rep.codes[y].clone()
        if self.mode=='exact_archive':self.archive_values=y.clone()
        return (x,x.clone()) if self.dual else (x,)
    def reset(self,state):
        if self.mode=='exact_archive' and self.archive_values is not None:self.archive_values.fill_(UNKNOWN)
        return tuple(torch.zeros_like(x) for x in state)
    def write(self,state,y,mask):
        code=self.rep.codes[y]
        if self.mode=='exact_archive':
            if self.archive_values is None:raise ValueError('No initialized archive')
            self.archive_values=torch.where(mask,y,self.archive_values)
        if self.mode.startswith('fast_slow'):
            return self.fastslow.write(*state,code,mask)
        return tuple(torch.where(mask[...,None],code,x) for x in state)
    def step(self,state):
        if self.mode=='recurrent': return (self.repair(state[0]),)
        if self.mode=='nearest_maintenance': return (self.rep.codes[self.rep.nearest(state[0])],)
        if self.mode in ('fast_slow','fast_slow_equal'):
            return self.fastslow(*state,equal=self.mode.endswith('equal'))
        return state
    def view(self,state):
        if self.mode=='state_off': return torch.zeros_like(state[0])
        if self.mode=='two_copy_mean': return (state[0]+state[1])/2
        z=state[0]
        if self.mode=='readout_repair': z=self.repair(z)
        elif self.mode=='recurrent_query4':
            for _ in range(4):z=self.repair(z)
        return z
    def predict(self,state):
        if self.mode=='exact_archive':
            if self.archive_values is None:raise ValueError('Explicit protected records not initialized')
            return self.archive_values.clone()
        if self.mode=='nearest_read':return self.rep.nearest(state[0])
        return self.rep(self.view(state)).argmax(-1)
    def active_state_bytes_per_stream(self):
        # Raw activation bytes only. Exact ledger is a separate intact control.
        return KEYS*D*4*(2 if self.dual else 1)
